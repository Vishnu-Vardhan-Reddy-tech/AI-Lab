from collections import deque

# -------- FIXED GRAPH DATA --------
vertices = ['A', 'B', 'C', 'D', 'E', 'F', 'I', 'G', 'H', 'K', 'L']
edges = [
    ('A','B'), ('A','C'), ('A','D'),
    ('B','E'), ('B','F'),
    ('D','I'),
    ('E','G'), ('E','H'),
    ('I','K'), ('I','L')
]

graph = {v: [] for v in vertices}
for u, v in edges:
    graph[u].append(v)

source = 'A'
goal = 'L'

# -------- INITIAL OUTPUT --------
print("Source Node:", source)
print("\nVertices (V):", vertices)
print("\nEdges (E):", edges)
print("\n-----------------------------\n")

# -------- BFS IMPLEMENTATION --------
queue = deque([source])
visited = []
parent = {source: None}
step = 1

while queue:
    current = queue.popleft()

    if current not in visited:
        visited.append(current)

    unvisited = [v for v in vertices if v not in visited]
    children = []

    for neighbor in graph[current]:
        if neighbor not in visited and neighbor not in queue:
            queue.append(neighbor)
            parent[neighbor] = current
            children.append(neighbor)

    print(f"Step {step}:")
    print("Visiting Node:", current)
    print("Visited Nodes:", visited)
    print("Unvisited Nodes:", unvisited if unvisited else 0)
    print("Child Vertices:", children)
    print()

    if current == goal:
        break

    step += 1

print("-----------------------------")

# -------- FINAL BFS ORDER --------
print("\nFinal Visited Nodes (BFS Order):", visited)
print("Non-Visited Nodes:", 0)

# -------- PATH RECONSTRUCTION --------
path = []
node = goal
while node is not None:
    path.append(node)
    node = parent[node]

path.reverse()
print("Path of A:", " → ".join(path))
