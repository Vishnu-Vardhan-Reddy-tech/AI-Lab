from collections import deque

# -------- FIXED GRAPH (WITH COSTS) --------
graph = {
    'S': [('A', 2), ('B', 5)],
    'A': [('C', 2), ('D', 4)],
    'B': [('D', 1)],
    'C': [('G', 3)],
    'D': [('G', 2)],
    'G': []
}

vertices = ['S', 'A', 'B', 'C', 'D', 'G']
edges = [
    ('S','A',2), ('S','B',5),
    ('A','C',2), ('A','D',4),
    ('B','D',1),
    ('C','G',3),
    ('D','G',2)
]

source = 'S'
goal = 'G'

# -------- INITIAL OUTPUT --------
print("Source Node:", source)
print("\nVertices (V):", vertices)
print("\nEdges (E):", edges)
print("\n-----------------------------\n")

# -------- BFS IMPLEMENTATION --------
queue = deque([source])
visited = []
parent = {source: None}
step = 1

while queue:
    current = queue.popleft()

    if current not in visited:
        visited.append(current)

    unvisited = [v for v in vertices if v not in visited]
    children = []

    for neighbor, cost in graph[current]:
        if neighbor not in visited and neighbor not in queue:
            queue.append(neighbor)
            parent[neighbor] = current
            children.append(neighbor)

    print(f"Step {step}:")
    print("Visiting Node:", current)
    print("Visited Nodes:", visited)
    print("Unvisited Nodes:", unvisited if unvisited else 0)
    print("Child Vertices:", children)
    print()

    if current == goal:
        break

    step += 1

print("-----------------------------")

# -------- FINAL BFS ORDER --------
print("\nFinal Visited Nodes (BFS Order):", visited)
print("Non-Visited Nodes:", 0)

# -------- PATH RECONSTRUCTION --------
path = []
node = goal
while node is not None:
    path.append(node)
    node = parent[node]

path.reverse()
print("Path of S:", " → ".join(path))
